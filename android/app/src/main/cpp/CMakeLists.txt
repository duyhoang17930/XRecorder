cmake_minimum_required(VERSION 3.22.1)
project(threeh_offline_asr LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
set(WHISPER_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(WHISPER_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(WHISPER_BUILD_SERVER OFF CACHE BOOL "" FORCE)
set(WHISPER_BUILD_IS_DEV OFF CACHE BOOL "" FORCE)
set(GGML_NATIVE OFF CACHE BOOL "" FORCE)
set(GGML_OPENMP OFF CACHE BOOL "" FORCE)
set(GGML_CPU_KLEIDIAI OFF CACHE BOOL "" FORCE)
set(GGML_AVX OFF CACHE BOOL "" FORCE)
set(GGML_AVX2 OFF CACHE BOOL "" FORCE)
set(GGML_FMA OFF CACHE BOOL "" FORCE)
set(GGML_F16C OFF CACHE BOOL "" FORCE)
set(GGML_CPU ON CACHE BOOL "" FORCE)
set(GGML_CPU_ALL_VARIANTS OFF CACHE BOOL "" FORCE)
# This distribution contains only Android CPU sources, not unused GPU/server SDKs.
foreach(backend BLAS CANN CUDA ET HIP METAL MUSA RPC VIRTGPU SYCL VULKAN WEBGPU ZDNN OPENCL HEXAGON ZENDNN OPENVINO)
  set(GGML_${backend} OFF CACHE BOOL "" FORCE)
endforeach()
foreach(backend COREML OPENVINO VITISAI)
  set(WHISPER_${backend} OFF CACHE BOOL "" FORCE)
endforeach()
add_subdirectory(third_party/whisper.cpp)
foreach(target whisper ggml-base ggml-cpu)
  target_compile_options(${target} PRIVATE -O3)
endforeach()
set(RNNOISE_DIR ${CMAKE_CURRENT_SOURCE_DIR}/third_party/rnnoise)
add_library(rnnoise STATIC
  ${RNNOISE_DIR}/src/denoise.c ${RNNOISE_DIR}/src/rnn.c
  ${RNNOISE_DIR}/src/rnn_data.c ${RNNOISE_DIR}/src/pitch.c
  ${RNNOISE_DIR}/src/kiss_fft.c ${RNNOISE_DIR}/src/celt_lpc.c)
target_include_directories(rnnoise PUBLIC ${RNNOISE_DIR}/include PRIVATE ${RNNOISE_DIR}/src)
target_compile_options(rnnoise PRIVATE -O3)
add_library(threeh_whisper SHARED whisper_jni.cpp)
target_link_libraries(threeh_whisper PRIVATE whisper rnnoise log android)
target_link_options(threeh_whisper PRIVATE "-Wl,-z,max-page-size=16384")
